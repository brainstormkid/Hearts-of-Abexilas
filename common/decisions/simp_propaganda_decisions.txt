simp_propaganda_category = {    #Always make sure your decisions are in a categroy, that you have set earlier.

simp_choose_poster_1 = {    #This is our 1st decison. It will set the 1st poster. The 1st poster is the 2nd frame of the image.
    icon = GFX_decision_generic_propaganda
    allowed = { always = yes }
    visible = { always = yes }
    available = { always = yes }
    cost = 25   #How much political Power it costs to take.

    complete_effect = { #We can put any effect (reward) here. Also, set our variables with the set_variable effect.
        if = {
            limit = {
                NOT = {
                    has_country_flag = active_simp_poster_1
                }
            }
            set_variable = {
                ROOT.simp_frame_to_set1 = 2 #We set the variable ROOT.simp_frame_to_set1 to 2. This is the frame we will use in the scripted_gui, to change the poster.
            }
            set_country_flag = active_simp_poster_1 #We set a country flag, so we can check if the player has already taken this decision, and not allow them to take it again.
        }
        else_if = {
            limit = {
                NOT = {
                    has_country_flag = active_simp_poster_2
                }
            }
            set_variable = {
                ROOT.simp_frame_to_set2 = 2
            }
            set_country_flag = active_simp_poster_2
        }
        else_if = {
            limit = {
                NOT = {
                    has_country_flag = active_simp_poster_3
                }
            }
            set_variable = {
                ROOT.simp_frame_to_set3 = 2
            }
            set_country_flag = active_simp_poster_3
        }
        simp_set_poster = yes
    }
}

    icon = GFX_decision_cancel
    allowed = { always = yes }
    visible = { always = yes }

                  check_variable = { ROOT.simp_slot_frame2 > 1 }
                  check_variable = { ROOT.simp_slot_frame3 > 1 } }

    complete_effect = {
        clr_country_flag = active_simp_poster_1

        simp_clear_poster = yes     #Our custom effect. Sets the frame to 1, the blank poster, visualy making it blank, clearing it for the player.
    }
