simp_choose_poster_1 = {    #This is our 1st decison. It will set the 1st poster. The 1st poster is the 2nd frame of the image.
    icon = GFX_decision_generic_propaganda
    allowed = { always = yes }
    visible = { always = yes }
    available = { always = yes }
    cost = 25   #How much political Power it costs to take.

    complete_effect = { #We can put any effect (reward) here. Also, set our variables with the set_variable effect.
        set_variable = { ROOT.simp_frame_to_set = 2 }   #Here, we set the poster. We set it to 2, because 1 is already the blank poster. 
        simp_set_poster = yes #Here, we make the variable we just set (simp_frame_to_set) to be equal to (simp_slot_frame), the one that actualy changes the frame. We do this with our custom effect, we just defined in the common/scripted_effects
    }
}
