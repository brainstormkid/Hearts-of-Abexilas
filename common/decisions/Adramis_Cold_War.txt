		icon = generic_operation
		days_mission_timeout = 293
		selectable_mission = no
		is_good = no
		priority = 10
		allowed = {
			OR = {
				original_tag = QYN
				original_tag = ZAM
				original_tag = GEH
			}
		}
		
		available = {
			always = no
		}

		timeout_effect = {
			add_war_support = -1
			set_country_flag = ACW_lost
			
			every_other_country = {
                limit = {
					OR = {
						original_tag = QYN
						original_tag = ZAM
						original_tag = GEH
					}
                    has_global_flag = ACW_ongoing
                    NOT = { has_country_flag = ACW_lost }
                    NOT = {
                        any_other_country = {
                            OR = {
								original_tag = QYN
								original_tag = ZAM
								original_tag = GEH
							}
                            NOT = { has_country_flag = ACW_lost }
                        }
                    }
                }
                remove_mission = ACW_incoming_collapse
                set_country_flag = ACW_won
                clr_global_flag = ACW_ongoing
                # winner's reward goes here
			}
		}
		
		cancel_trigger = {
			OR = {
				AND = {
					GEH = { has_country_flag = ACW_lost }
					QYN = { has_country_flag = ACW_lost }
				}
				AND = {
					GEH = { has_country_flag = ACW_lost }
					ZAM = { has_country_flag = ACW_lost }
				}
				AND = {
					ZAM = { has_country_flag = ACW_lost }
					QYN = { has_country_flag = ACW_lost }
				}
			}
		}
		
	}
